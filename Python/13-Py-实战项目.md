### 13.1 完整源码

```python
# todo.py —— 命令行待办清单
import json
from pathlib import Path
from datetime import datetime

DATA_FILE = Path("todos.json")

class Task:
    """待办任务"""
    def __init__(self, title, done=False):
        self.title = title
        self.done = done
        self.created_at = datetime.now().strftime("%Y-%m-%d %H:%M")

    def to_dict(self):
        return {"title": self.title, "done": self.done, "created_at": self.created_at}

    @classmethod
    def from_dict(cls, data):
        task = cls(data["title"], data["done"])
        task.created_at = data.get("created_at", "未知")
        return task

    def __str__(self):
        status = "[✓]" if self.done else "[ ]"
        return f"{status} {self.title}（创建于 {self.created_at}）"


class TodoManager:
    """任务管理器：负责存储与操作"""
    def __init__(self):
        self.tasks = self._load()

    def _load(self):
        """从文件加载任务，文件不存在返回空列表"""
        if not DATA_FILE.exists():
            return []
        try:
            with open(DATA_FILE, "r", encoding="utf-8") as f:
                data = json.load(f)
            return [Task.from_dict(item) for item in data]
        except (json.JSONDecodeError, ValueError):
            print("警告：数据文件损坏，已重置")
            return []

    def _save(self):
        """保存任务到文件"""
        with open(DATA_FILE, "w", encoding="utf-8") as f:
            json.dump([t.to_dict() for t in self.tasks], f, ensure_ascii=False, indent=2)

    def add(self, title):
        if not title.strip():
            raise ValueError("任务内容不能为空")
        self.tasks.append(Task(title))
        self._save()
        print(f"已添加：{title}")

    def list_all(self):
        if not self.tasks:
            print("暂无任务，用 add 添加一个吧！")
            return
        print(f"共 {len(self.tasks)} 个任务：")
        for i, task in enumerate(self.tasks, start=1):
            print(f"  {i}. {task}")

    def remove(self, index):
        task = self._get_task(index)
        self.tasks.pop(index - 1)
        self._save()
        print(f"已删除：{task.title}")

    def mark_done(self, index):
        task = self._get_task(index)
        task.done = True
        self._save()
        print(f"已完成：{task.title}")

    def _get_task(self, index):
        """按序号取任务，带边界检查"""
        if index < 1 or index > len(self.tasks):
            raise IndexError(f"序号无效：{index}（范围 1-{len(self.tasks)}）")
        return self.tasks[index - 1]


def command_loop():
    """命令行交互主循环"""
    manager = TodoManager()

    while True:
        print("\n命令：list | add 任务 | done 序号 | rm 序号 | quit")
        cmd = input("> ").strip().lower()

        try:
            if cmd == "quit":
                print("再见！")
                break
            elif cmd == "list":
                manager.list_all()
            elif cmd.startswith("add "):
                manager.add(cmd[4:])
            elif cmd.startswith("done "):
                manager.mark_done(int(cmd[5:]))
            elif cmd.startswith("rm "):
                manager.remove(int(cmd[3:]))
            else:
                print("未知命令，请重试")
        except (ValueError, IndexError) as e:
            print(f"操作失败：{e}")
        except Exception as e:
            print(f"发生未知错误：{e}")


if __name__ == "__main__":
    command_loop()
```

### 13.4 简单测试

```python
# test_todo.py —— 用临时文件测试核心逻辑
import json
import tempfile
from pathlib import Path


def test_task_roundtrip():
    """测试 Task 的序列化和反序列化"""
    from todo import Task
    task = Task("测试任务", done=True)
    restored = Task.from_dict(task.to_dict())
    assert restored.title == "测试任务"
    assert restored.done is True
    print("test_task_roundtrip: PASS")


def test_add_save_load(tmp_path):
    """测试添加任务后能保存并重新加载"""
    from todo import Task
    # 用临时目录模拟（真实项目会做依赖注入，这里演示思路）
    tasks = [Task("任务A"), Task("任务B")]
    data = [t.to_dict() for t in tasks]

    f = tmp_path / "todos.json"
    with open(f, "w", encoding="utf-8") as fp:
        json.dump(data, fp, ensure_ascii=False)

    with open(f, "r", encoding="utf-8") as fp:
        loaded = json.load(fp)
    assert len(loaded) == 2
    assert loaded[0]["title"] == "任务A"
    print("test_add_save_load: PASS")


def run_all():
    import tempfile
    with tempfile.TemporaryDirectory() as td:
        test_task_roundtrip()
        test_add_save_load(Path(td))


if __name__ == "__main__":
    run_all()
```

### 13.6 这个项目教你什么

- **模块化思维**：先设计 `Task`（数据）和 `TodoManager`（操作）分层
- **健壮性**：每个用户输入都可能出错，`try/except` 接住所有边界
- **数据持久化**：JSON 存盘，重启不丢
- **可测试性**：核心逻辑与交互分离，才能写单元测试
